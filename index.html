      CREATE TABLE IF NOT EXISTS pt_daily_records (
    date TEXT PRIMARY KEY,
    rows_data JSONB NOT NULL DEFAULT '[]'::jsonb,
    total_count INTEGER DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);
ALTER TABLE pt_daily_records ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public all" ON pt_daily_records FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
